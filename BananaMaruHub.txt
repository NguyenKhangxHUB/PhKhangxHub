-- 🍎 "key chỗ này" -> Thay key thực tế ở đây
local VALID_KEY = "GOKAI-2026-KEY"
-- 🍎 "link gét key chỗ này" -> Link lấy key
local GET_KEY_LINK = "https://link4m.net/qC4a8p"

local StorageFile = "keyauth_data.json" -- file lưu thời gian xác thực

-- Hàm kiểm tra nếu đã xác thực trong 24h qua
local function isStillValid()
    local success, content = pcall(readfile, StorageFile)
    if not success then return false end
    local data = game:GetService("HttpService"):JSONDecode(content)
    if data and data.timestamp then
        return (os.time() - data.timestamp) < 86400 -- 24 giờ = 86400 giây
    end
    return false
end

-- Nếu đã xác thực, tự động chạy script chính và không hiện GUI
if isStillValid() then
    loadstring(game:HttpGet("https://raw.githubusercontent.com/NguyenKhangxHUB/PhKhangxHub/refs/heads/main/3580340031672416.lua"))()
    return
end

-- Hàm rung frame khi sai key
local function shakeFrame()
    local origPos = mainFrame.Position
    local shakeOffset = 5
    local tweenInfo = TweenInfo.new(0.05, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
    local tweens = {}
    local offsets = {UDim2.new(0, shakeOffset, 0, 0), UDim2.new(0, -shakeOffset, 0, 0), UDim2.new(0, 0, 0, shakeOffset), UDim2.new(0, 0, 0, -shakeOffset)}
    for i, off in ipairs(offsets) do
        local goal = {Position = origPos + off}
        local tween = game:GetService("TweenService"):Create(mainFrame, tweenInfo, goal)
        table.insert(tweens, tween)
    end
    -- Phát tuần tự
    local seq = function()
        for _, t in ipairs(tweens) do
            t:Play()
            t.Completed:Wait()
        end
        -- Quay về vị trí cũ
        local tweenBack = game:GetService("TweenService"):Create(mainFrame, TweenInfo.new(0.05), {Position = origPos})
        tweenBack:Play()
    end
    spawn(seq)
end

-- Animation biến mất (kéo sang trái và mờ dần)
local function fadeOutAndRemove()
    local goalPos = mainFrame.Position - UDim2.new(0, 500, 0, 0)
    local tween1 = game:GetService("TweenService"):Create(mainFrame, TweenInfo.new(0.5, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {Position = goalPos})
    local tween2 = game:GetService("TweenService"):Create(mainFrame, TweenInfo.new(0.5, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {BackgroundTransparency = 1})
    -- Làm mờ tất cả con
    for _, child in ipairs(mainFrame:GetDescendants()) do
        if child:IsA("GuiObject") and child ~= mainFrame then
            spawn(function()
                local t = game:GetService("TweenService"):Create(child, TweenInfo.new(0.5), {TextTransparency = 1, BackgroundTransparency = 1, ImageTransparency = 1})
                t:Play()
            end)
        end
    end
    tween1:Play()
    tween2:Play()
    tween1.Completed:Wait()
    gui:Destroy()
end

-- Animation xuất hiện (scale từ nhỏ đến lớn)
local function introAnimation()
    mainFrame.Size = UDim2.new(0, 0, 0, 0)
    mainFrame.Position = UDim2.new(0.5, 0, 0.5, 0)
    local tweenSize = game:GetService("TweenService"):Create(mainFrame, TweenInfo.new(0.4, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Size = UDim2.new(0, 300, 0, 220)})
    local tweenPos = game:GetService("TweenService"):Create(mainFrame, TweenInfo.new(0.4, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Position = UDim2.new(0.5, -150, 0.5, -110)})
    tweenSize:Play()
    tweenPos:Play()
end

-- Gọi intro
introAnimation()

-- Xử lý nút Check Key
checkBtn.MouseButton1Click:Connect(function()
    local inputKey = keyInput.Text
    if inputKey == VALID_KEY then
        -- Lưu timestamp xác thực
        local data = {timestamp = os.time()}
        pcall(function()
            writefile(StorageFile, game:GetService("HttpService"):JSONEncode(data))
        end)
        notiLabel.Text = "Key đúng! Đang kích hoạt script..."
        wait(0.5)
        fadeOutAndRemove()
        -- Chạy script chính
        loadstring(game:HttpGet("https://raw.githubusercontent.com/NguyenKhangxHUB/PhKhangxHub/refs/heads/main/3580340031672416.lua"))()
    else
        notiLabel.Text = "Sai key! Vui lòng thử lại."
        shakeFrame()
        wait(1.5)
        notiLabel.Text = "Key đúng hiệu lực 24h"
    end
end)
