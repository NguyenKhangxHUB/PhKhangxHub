    "ADMINFIGHT",
    "GIFTING_HOURS",

    "SUB2GAMERROBOT_RESET1",
    "CHANDLER",
    "FUDD10",
    "FUDD10_V2",
    "BIGNEWS",
    "THEGREATACE",
    "SUB2CAPTAINMAUI",
    "KITTGAMING",
    "ENYU_IS_PRO",
    "MAGICBUS",
    "JCWK",
    "STARCODEHEO",
    "BLUXXY",
    "SUB2OFFICIALNOOBIE",
    "STRAWHATMAINE",
    "TANTAIGAMING",
    "AXIORE",
    "SUB2DAIGROCK",
    "SUB2FER999"
}

task.spawn(function()
    local ReplicatedStorage = game:GetService("ReplicatedStorage")
    local Remote = ReplicatedStorage:WaitForChild("Remotes"):WaitForChild("Redeem")

    for _, code in ipairs(Codes) do
        pcall(function()
            Remote:InvokeServer(code)
        end)
        task.wait(0.3)
    end
end)
